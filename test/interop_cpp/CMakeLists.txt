cmake_minimum_required(VERSION 3.8)
project(rclsuru_interop_cpp)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(tf2_msgs REQUIRED)
foreach(node cpp_talker cpp_listener)
  add_executable(${node} src/${node}.cpp)
  ament_target_dependencies(${node} rclcpp std_msgs)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
add_executable(cpp_msgs src/cpp_msgs.cpp)
ament_target_dependencies(cpp_msgs rclcpp std_msgs geometry_msgs sensor_msgs)
install(TARGETS cpp_msgs DESTINATION lib/${PROJECT_NAME})
foreach(node cpp_set_bool_server cpp_set_bool_client)
  add_executable(${node} src/${node}.cpp)
  ament_target_dependencies(${node} rclcpp std_srvs)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
foreach(node cpp_lookup_server cpp_lookup_client)
  add_executable(${node} src/${node}.cpp)
  ament_target_dependencies(${node} rclcpp rclcpp_action tf2_msgs)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
ament_package()
